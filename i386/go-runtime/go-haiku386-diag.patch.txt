Opt-in diagnostics for the Haiku/386 startup hang. NOT applied by build.sh.

Apply by hand when investigating, then set GOHAIKUSTALLPANIC=<seconds>:
sysmon notices that no P has scheduled anything for that long and throws,
printing the scheduler's own view first - per-P status and idle-list
membership, and per-m wakeup accounting, spinning/blocked/locks state, the
in-flight libc call, and any runtime mutex an m is asleep on inside lock2.

This is the only way to see the hang: Haiku's strace will not attach to a
running team, SIGQUIT to a wedged process produces nothing, and a watchdog
goroutine cannot run when no goroutine can be scheduled at all. sysmon can,
because it runs without a P.

What it established, and what it ruled out, is written up in AGENTS.md.

+// checkHaikuStall reports the scheduler's own view when nothing has been
+// scheduled for GOHAIKUSTALLPANIC seconds, then throws so every goroutine stack
+// is dumped.
+//
+// sysmon is the only place this can be done: it runs without a P, and in this
+// hang no goroutine can run at all. Haiku's strace will not attach to a live
+// team and SIGQUIT to a wedged process produces nothing.
+//
+// Diagnostic only; off unless GOHAIKUSTALLPANIC is set.
+var (
+	haikuStallNS      int64
+	haikuStallOnce    bool
+	haikuLastProgress int64
+	haikuLastTickSum  int64
+)
+
+func checkHaikuStall(now int64) {
+	if !haikuStallOnce {
+		haikuStallOnce = true
+		v := gogetenv("GOHAIKUSTALLPANIC")
+		n := int64(0)
+		for i := 0; i < len(v); i++ {
+			c := v[i]
+			if c < '0' || c > '9' {
+				n = 0
+				break
+			}
+			n = n*10 + int64(c-'0')
+		}
+		haikuStallNS = n * 1e9
+	}
+	if haikuStallNS == 0 {
+		return
+	}
+	tickSum := int64(0)
+	for _, pp := range allp {
+		if pp == nil {
+			continue
+		}
+		tickSum += int64(pp.schedtick)
+	}
+	if tickSum != haikuLastTickSum || haikuLastProgress == 0 {
+		haikuLastTickSum = tickSum
+		haikuLastProgress = now
+		return
+	}
+	if now-haikuLastProgress <= haikuStallNS {
+		return
+	}
+	print("haiku: stalled ", (now-haikuLastProgress)/1e6, "ms globrunq=", sched.runq.size,
+		" npidle=", sched.npidle.Load(), " nmidle=", sched.nmidle,
+		" nmspinning=", sched.nmspinning.Load(), " needspinning=", sched.needspinning.Load(), "\n")
+	// Walk the idle list so a P that is _Pidle yet absent from it is visible.
+	onIdle := 0
+	for l := sched.pidle.ptr(); l != nil; l = l.link.ptr() {
+		onIdle++
+	}
+	print("haiku: pidle list length=", onIdle, "\n")
+	for _, pp := range allp {
+		if pp == nil {
+			continue
+		}
+		listed := 0
+		for l := sched.pidle.ptr(); l != nil; l = l.link.ptr() {
+			if l == pp {
+				listed = 1
+				break
+			}
+		}
+		print("haiku: P", pp.id, " status=", pp.status, " schedtick=", pp.schedtick,
+			" runq=", int32(pp.runqtail-pp.runqhead), " onIdleList=", listed,
+			" m=", pp.m != 0, "\n")
+	}
+	// Per-m wakeup accounting. A misdirected unlock2 post shows up here as an m
+	// whose posts run ahead of its takes while some OTHER m waits forever: the
+	// wakeup was delivered to a thread that was not waiting for it.
+	for mp := allm; mp != nil; mp = mp.alllink {
+		// mp.libcall records an in-flight libc call on this port, so a non-zero
+		// fn means this m is inside libc right now. That is the difference
+		// between "stuck in Go code" and "stuck in a Haiku libc call while
+		// holding a runtime lock".
+		print("haiku: m", mp.id, " posts=", mp.semaPosts.Load(), " takes=", mp.semaTakes.Load(),
+			" spinning=", mp.spinning, " blocked=", mp.blocked, " locks=", mp.locks,
+			" hasP=", mp.p != 0, " curg=", mp.curg != nil,
+			" libcall.fn=", hex(mp.libcall.fn), " libcall.n=", mp.libcall.n)
+		if lp := mp.haikuWaitLock; lp != 0 {
+			key := atomic.Loaduintptr(&(*mutex)(unsafe.Pointer(lp)).key)
+			print(" waitLock=", hex(lp), " key=", hex(key),
+				" locked=", key&0x001 != 0, " sleeping=", key&0x002 != 0,
+				" spinning=", key&0x100 != 0, " waiters=", key&^uintptr(0x3FF) != 0)
+		}
+		print("\n")
+	}
+	print("haiku: libc snooze=", hex(uintptr(unsafe.Pointer(&libc_snooze))),
+		" usleep=", hex(uintptr(unsafe.Pointer(&libc_usleep))),
+		" select=", hex(uintptr(unsafe.Pointer(&libc_select))),
+		" malloc=", hex(uintptr(unsafe.Pointer(&libc_malloc))),
+		" read=", hex(uintptr(unsafe.Pointer(&libc_read))),
+		" sem_init=", hex(uintptr(unsafe.Pointer(&libc_sem_init))), "\n")
+	print("haiku: libc sem_post=", hex(uintptr(unsafe.Pointer(&libc_sem_post))),
+		" sem_wait=", hex(uintptr(unsafe.Pointer(&libc_sem_wait))),
+		" sem_timedwait=", hex(uintptr(unsafe.Pointer(&libc_sem_timedwait))),
+		" clock_gettime=", hex(uintptr(unsafe.Pointer(&libc_clock_gettime))),
